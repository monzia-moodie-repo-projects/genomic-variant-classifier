    n = len(atoms)
    radii = [_VDW_RADII.get(a["element"], _DEFAULT_VDW) + _WATER_PROBE for a in atoms]
    xs = [a["x"] for a in atoms]
    ys = [a["y"] for a in atoms]
    zs = [a["z"] for a in atoms]

    sasa_res: dict[int, float] = {}
    for i in range(n):
        ri = radii[i]
        xi, yi, zi = xs[i], ys[i], zs[i]
        neigh = []
        for j in range(n):
            if j == i:
                continue
            cutoff = ri + radii[j]
            dx = xi - xs[j]
            dy = yi - ys[j]
            dz = zi - zs[j]
            if dx * dx + dy * dy + dz * dz < cutoff * cutoff:
                neigh.append(j)
        accessible = 0
        for (sx, sy, sz) in _SPHERE:
            px = xi + ri * sx
            py = yi + ri * sy
            pz = zi + ri * sz
            buried = False
            for j in neigh:
                dx = px - xs[j]
                dy = py - ys[j]
                dz = pz - zs[j]
                if dx * dx + dy * dy + dz * dz < radii[j] * radii[j]:
                    buried = True
                    break
            if not buried:
                accessible += 1
        area = 4.0 * math.pi * ri * ri * accessible / float(len(_SPHERE))
        seq = atoms[i]["seq_id"]
        sasa_res[seq] = sasa_res.get(seq, 0.0) + area
