329:     """
330:     Compute per-residue relative solvent accessibility (RSA) in [0, 1] via a
331:     Shrake-Rupley SASA over ALL atoms, normalised by Tien et al. max-ASA and
332:     clamped to [0, 1] (see _RSA_CLAMP_NOTE).
333: 
334:     Raises CIFParseError if any residue's RAW RSA exceeds _RSA_FAIL_LOUD_MAX or is
335:     negative -- that indicates a real geometry failure, not a terminal artefact,
336:     and must fail loud rather than be silently clamped.
337:     """
338:     n = len(atoms)
339:     radii = [_VDW_RADII.get(a["element"], _DEFAULT_VDW) + _WATER_PROBE for a in atoms]
340:     xs = [a["x"] for a in atoms]
341:     ys = [a["y"] for a in atoms]
342:     zs = [a["z"] for a in atoms]
343: 
344:     sasa_res: dict[int, float] = {}
345:     for i in range(n):
346:         ri = radii[i]
347:         xi, yi, zi = xs[i], ys[i], zs[i]
348:         neigh = []
349:         for j in range(n):
350:             if j == i:
351:                 continue
352:             cutoff = ri + radii[j]
353:             dx = xi - xs[j]
354:             dy = yi - ys[j]
355:             dz = zi - zs[j]
356:             if dx * dx + dy * dy + dz * dz < cutoff * cutoff:
357:                 neigh.append(j)
358:         accessible = 0
359:         for (sx, sy, sz) in _SPHERE:
360:             px = xi + ri * sx
361:             py = yi + ri * sy
362:             pz = zi + ri * sz
363:             buried = False
364:             for j in neigh:
365:                 dx = px - xs[j]
366:                 dy = py - ys[j]
367:                 dz = pz - zs[j]
368:                 if dx * dx + dy * dy + dz * dz < radii[j] * radii[j]:
369:                     buried = True
370:                     break
371:             if not buried:
372:                 accessible += 1
373:         area = 4.0 * math.pi * ri * ri * accessible / float(len(_SPHERE))
374:         seq = atoms[i]["seq_id"]
375:         sasa_res[seq] = sasa_res.get(seq, 0.0) + area
376: 
377:     comp = per_residue_comp(atoms)
378:     rsa: dict[int, float] = {}
379:     for seq, sasa in sasa_res.items():
380:         max_asa = _MAX_ASA.get(comp.get(seq, ""), None)
381:         if not max_asa:
382:             rsa[seq] = DEFAULT_RSA
383:             continue
384:         raw = sasa / max_asa
385:         if raw < 0.0 or raw > _RSA_FAIL_LOUD_MAX:
386:             raise CIFParseError(
387:                 f"residue {seq} RSA={raw:.3f} outside [0,{_RSA_FAIL_LOUD_MAX}] "
388:                 f"-- geometry failure, not a terminal artefact"
389:             )
390:         rsa[seq] = min(1.0, max(0.0, raw))  # clamp; see _RSA_CLAMP_NOTE
391:     return rsa
