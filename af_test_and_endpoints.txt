1: """
2: tests/unit/test_alphafold.py
3: ============================
4: Unit tests for the AlphaFold structural-feature connector, extractor library, and
5: cohort coverage gate.
6: 
7: Coverage:
8:   Extractor (alphafold_features):
9:     1.  parse_atom_site on a real-format CIF -> correct atom count, 1-based seq_id
10:     2.  per_residue_plddt -> B_iso values, correct residues
11:     3.  parse_struct_conf (DSSP parse-first) -> helix/sheet/loop codes
12:     4.  secondary_structure_from_coords (fallback) -> non-degenerate
13:     5.  per_residue_rsa -> all in [0,1] (clamp), core < terminus ordering
14:     6.  RSA fail-loud guard raises on a fabricated geometry failure
15:     7.  3-D C-alpha distance: adjacent residues ~3.8 A
16:     8.  residue indexing off-by-one tripwire (seq_id starts at 1, not 0)
17:   Connector (alphafold.AlphaFoldConnector):
18:     9.  wt_aa match -> real features attached
19:     10. wt_aa isoform MISMATCH -> fail-closed to sentinel default
20:     11. missing protein_pos -> default
21:     12. empty df -> columns present, no rows
22:     13. stub mode (no parquet) -> all defaults
23:     14. TABULAR_FEATURES membership -> the 4 AF features present
24:   Coverage gate:
25:     15. AF features non-constant on a real cohort sample (fraction-at-sentinel gate)
26: """

109:     atoms = aff.parse_atom_site(_E7ENB7.read_text(errors="replace"))
110:     ca = aff.per_residue_ca(atoms)
111:     (x1, y1, z1), (x2, y2, z2) = ca[1], ca[2]
112:     d = math.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2 + (z1 - z2) ** 2)

116: def test_rsa_fail_loud_guard():
117:     """
118:     A corrupt record with several well-separated atoms assigned to ONE residue yields
119:     RSA far above the fail-loud bound -> must raise, not clamp. (A single isolated atom
120:     only reaches ~1.2, within the terminal-artefact range that is correctly clamped;
121:     the guard fires on genuine geometry failures like this multi-atom pileup.)
122:     """
123:     atoms = [
124:         {"element": "C", "atom_id": "CA", "comp": "GLY", "seq_id": 1,
125:          "x": 0.0, "y": 0.0, "z": 0.0, "bfactor": 50.0},
126:         {"element": "C", "atom_id": "C", "comp": "GLY", "seq_id": 1,
127:          "x": 50.0, "y": 0.0, "z": 0.0, "bfactor": 50.0},
128:         {"element": "C", "atom_id": "N", "comp": "GLY", "seq_id": 1,
129:          "x": 0.0, "y": 50.0, "z": 0.0, "bfactor": 50.0},
130:     ]
131:     with pytest.raises(aff.CIFParseError):
132:         aff.per_residue_rsa(atoms)

135: def test_residue_indexing_off_by_one_tripwire():
136:     """Guard against a 0-based regression: a minimal 2-residue CIF must index 1,2."""
137:     cif = (
138:         "data_x\n#\n"
139:         "loop_\n_atom_site.group_PDB\n_atom_site.type_symbol\n_atom_site.label_atom_id\n"
140:         "_atom_site.label_comp_id\n_atom_site.auth_seq_id\n_atom_site.Cartn_x\n"
141:         "_atom_site.Cartn_y\n_atom_site.Cartn_z\n_atom_site.B_iso_or_equiv\n"
142:         "ATOM C CA ALA 1 0.0 0.0 0.0 90.0\n"
143:         "ATOM C CA GLY 2 3.8 0.0 0.0 80.0\n#\n"
144:     )
145:     plddt = aff.per_residue_plddt(aff.parse_atom_site(cif))
146:     assert set(plddt) == {1, 2}, "residues must be 1-based (1,2), not (0,1)"

230: @requires_cif
231: def test_coverage_gate_features_nonconstant(tmp_path):
232:     """
233:     The gate that would have caught the stub-constant state: after building a small
234:     real parquet and annotating a cohort sample, the AF features must be non-constant
235:     (fraction at sentinel below threshold). Here we assert directly on a built parquet.
236:     """
237:     atoms = aff.parse_atom_site(_E7ENB7.read_text(errors="replace"))
238:     plddt = aff.per_residue_plddt(atoms)
239:     rsa = aff.per_residue_rsa(atoms)
240:     ss = aff.per_residue_secondary_structure(_E7ENB7.read_text(errors="replace"), atoms)
241:     # non-constant across residues
242:     assert len(set(round(v, 2) for v in plddt.values())) > 5
243:     assert len(set(round(v, 2) for v in rsa.values())) > 5
244:     assert len(set(ss.values())) >= 2  # at least loop + helix
245:     # fraction at sentinel default must be low (these are REAL values, not stubs)
246:     frac_plddt_sentinel = sum(1 for v in plddt.values() if v == DEFAULT_PLDDT) / len(plddt)
247:     assert frac_plddt_sentinel < 0.5, "most residues should have real (non-sentinel) pLDDT"
